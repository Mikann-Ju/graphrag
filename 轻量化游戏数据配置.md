import pandas as pd

import random

class LightweightGameDataGenerator:
    """轻量化游戏数据生成器 - 适配MacBook Pro 2020"""

        random.seed(seed)
        np.random.seed(seed)
        
        # 减少数据规模
        self.game_types = ["消消乐", "纸牌"]  # 减少到2种
        self.room_types = ["免费场", "付费场", "高级付费场"]

    def generate_lightweight_data(self, output_dir: str = "lightweight_game_data"):
        """生成轻量化数据"""

        payments_df = self.generate_payment_records(sessions_df)
        
        # 生成精简文档
        documents_df = self.generate_compact_documents(
            players_df, rooms_df, sessions_df, payments_df
        )
        
        # 保存数据
        documents_df.to_csv(f"{output_dir}/input/documents.csv", index=False, encoding='utf-8')
        
        # 生成统计
        stats = {
            "数据生成时间": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
            "玩家总数": len(players_df),
            "场次总数": len(rooms_df), 
            "游戏会话总数": len(sessions_df),
            "文档总数": len(documents_df),
            "预计磁盘使用": "< 100MB",
            "预计内存需求": "< 4GB",
            "适配硬件": "MacBook Pro 2020"
        }
        
        with open(f"{output_dir}/stats.json", 'w', encoding='utf-8') as f:
            json.dump(stats, f, ensure_ascii=False, indent=2)
        
        print(f"✅ 轻量化数据生成完成！")
        print(f"📁 输出目录: {output_dir}/")
        print(f"📄 文档数量: {len(documents_df)}")
        print(f"💾 磁盘使用: < 100MB")
        print(f"🧠 内存需求: < 4GB")
        
        return stats
    
    def generate_compact_documents(self, players_df, rooms_df, sessions_df, payments_df):
        """生成精简文档"""
        documents = []
        doc_id = 1
        
        # 只生成核心文档类型
        
        # 1. 玩家摘要文档（每10个玩家一个文档）
        for i in range(0, len(players_df), 10):
            batch = players_df.iloc[i:i+10]
            
            text = f"玩家群组分析 {doc_id}\n\n"
            for _, player in batch.iterrows():
                text += f"玩家{player['player_id']}：{player['segment']}类型，"
                text += f"总消费{player['total_spending']}元，"
                text += f"流失风险{player['churn_risk_score']}。\n"
            
            documents.append({
                "id": f"player_batch_{doc_id:03d}",
                "text": text.strip(),
                "metadata": json.dumps({
                    "document_type": "player_batch",
                    "batch_size": len(batch)
                })
            })
            doc_id += 1
        
        # 2. 场次汇总文档
        for room_type in rooms_df['room_type'].unique():
            rooms_subset = rooms_df[rooms_df['room_type'] == room_type]
            room_sessions = sessions_df[sessions_df['room_id'].isin(rooms_subset['room_id'])]
            
            if len(room_sessions) > 0:
                text = f"{room_type}分析报告\n\n"
                text += f"场次数量：{len(rooms_subset)}个\n"
                text += f"总参与：{len(room_sessions)}次\n"
                text += f"平均满意度：{room_sessions['satisfaction_rating'].mean():.2f}\n"
                text += f"平均ROI：{room_sessions['roi_ratio'].mean():.2f}\n"
                
                if room_type == "高级付费场":
                    text += "该类场次为高价值场次，建议重点优化。"
                elif room_type == "免费场":
                    text += "该类场次用于吸引新玩家，关注转化率。"
                else:
                    text += "该类场次为核心收益场次，平衡体验和收益。"
                
                documents.append({
                    "id": f"room_type_{doc_id:03d}",
                    "text": text,
                    "metadata": json.dumps({
                        "document_type": "room_analysis",
                        "room_type": room_type
                    })
                })
                doc_id += 1
        
        # 3. 整体业务洞察文档
        high_value_players = players_df[players_df['total_spending'] > players_df['total_spending'].quantile(0.8)]
        high_risk_players = players_df[players_df['churn_risk_score'] > 0.7]
        
        insight_text = f"""游戏运营核心洞察
        
        玩家概况：
        - 总玩家数：{len(players_df)}
        - 高价值玩家：{len(high_value_players)}人（{len(high_value_players)/len(players_df)*100:.1f}%）
        - 高流失风险：{len(high_risk_players)}人（{len(high_risk_players)/len(players_df)*100:.1f}%）
        
        收益分析：
        - 总收入：{payments_df['actual_amount'].sum():.2f}元
        - 人均消费：{payments_df['actual_amount'].sum()/len(players_df):.2f}元
        - 付费转化率：{len(payments_df)/len(sessions_df)*100:.1f}%
        
        运营建议：
        1. 重点关注高流失风险玩家的挽回
        2. 优化付费场次的投资回报率
        3. 加强新手玩家的引导和转化
        """
        
        documents.append({
            "id": f"business_insights_{doc_id:03d}",
            "text": insight_text.strip(),
            "metadata": json.dumps({
                "document_type": "business_insights",
                "total_players": len(players_df),
                "total_revenue": float(payments_df['actual_amount'].sum())
            })
        })
        
        return pd.DataFrame(documents)

import psutil
import time
import json

def monitor_graphrag_performance():
    """监控GraphRAG运行性能"""
    
    print("🔍 MacBook Pro 2020 性能监控")
    print("-" * 40)
    
    # 系统信息
    cpu_count = psutil.cpu_count()
    memory = psutil.virtual_memory()
    disk = psutil.disk_usage('.')
    
    print(f"CPU核心数: {cpu_count}")
    print(f"总内存: {memory.total / 1024**3:.1f}GB")
    print(f"可用内存: {memory.available / 1024**3:.1f}GB")
    print(f"磁盘使用: {disk.used / 1024**3:.1f}GB / {disk.total / 1024**3:.1f}GB")
    
    # 建议配置
    recommended_config = {
        "concurrent_requests": min(2, cpu_count),
        "max_memory_usage": f"{int(memory.available / 1024**3 * 0.6)}GB",
        "chunk_size": 400 if memory.total < 16 * 1024**3 else 800,
        "batch_size": 10 if memory.total < 16 * 1024**3 else 50
    }
    
    print("\n💡 推荐配置:")
    for key, value in recommended_config.items():
        print(f"  {key}: {value}")
    
    return recommended_config

## 故障排除

### 常见问题
1. **内存不足**：减少并发请求数到1
2. **磁盘空间不足**：删除快照和缓存
3. **响应慢**：使用gpt-3.5-turbo替代gpt-4
4. **索引失败**：检查API密钥和网络连接

### 应急方案
```bash
# 最小化配置
echo "concurrent_requests: 1" >> settings.yml
echo "batch_size: 5" >> settings.yml

# 清理缓存
rm -rf cache/
rm -rf .cache/

# 重新索引
graphrag index --root . --verbose
```
